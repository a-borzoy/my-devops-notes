# 1. Берем официальный образ Node.js (версия 18)
FROM node:18

# 2. Создаем рабочую папку внутри контейнера
WORKDIR /app

# 3. Копируем файлы package.json и package-lock.json для установки зависимостей
COPY package*.json ./

# 4. Устанавливаем библиотеки
RUN npm install

# 5. Копируем оставшийся код приложения
COPY . .

# 6. Говорим, что приложение работает на 3000 порту
EXPOSE 3000

# 7. Команда для старта приложения
CMD ["node", "server.js"]
